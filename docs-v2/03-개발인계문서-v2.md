## 4. Claude API 연동 코드

### 4.1 서버 구현 (Next.js App Router)

```typescript
// app/api/analyze/route.ts
import Anthropic from '@anthropic-ai/sdk';
import { getRegulations } from '@/lib/regulations';
import { buildSystemPrompt, RESULT_SCHEMA } from '@/lib/prompts';
import { saveAnalysis } from '@/lib/db';

const client = new Anthropic({
  apiKey: process.env.ANTHROPIC_API_KEY
});

export async function POST(req: Request) {
  const { lot, buildingUse, options } = await req.json();

  // 1. 해당 지자체 법규 조회 (DB)
  const regulations = await getRegulations(
    lot.regionCode,
    lot.용도지역
  );

  // 2. System Prompt 조립
  const systemPrompt = buildSystemPrompt(regulations, lot);

  // 3. SSE 스트리밍 응답
  const stream = new TransformStream();
  const writer = stream.writable.getWriter();
  const encoder = new TextEncoder();

  const sendEvent = async (event: string, data: any) => {
    await writer.write(
      encoder.encode(`event: ${event}\ndata: ${JSON.stringify(data)}\n\n`)
    );
  };

  // 비동기 분석 실행
  (async () => {
    try {
      // Step 0: 필지 정보 확인
      await sendEvent('step', { step: 0, status: 'done' });

      // Step 1: 법규 수집
      await sendEvent('step', { step: 1, status: 'processing' });
      // regulations는 이미 조회됨
      await sendEvent('step', { step: 1, status: 'done' });

      // Step 2: Claude AI 분석
      await sendEvent('step', { step: 2, status: 'processing' });

      const response = await client.messages.create({
        model: "claude-sonnet-4-20250514",
        max_tokens: 4096,
        system: systemPrompt,
        messages: [{
          role: "user",
          content: JSON.stringify({
            task: "신축 최대 규모 산출",
            lot, buildingUse, options,
            regulations: regulations.summary
          })
        }],
        tools: [{
          name: "building_feasibility_result",
          description: "신축규모검토 결과를 구조화된 JSON으로 반환",
          input_schema: RESULT_SCHEMA
        }],
        tool_choice: { type: "tool", name: "building_feasibility_result" }
      });

      await sendEvent('step', { step: 2, status: 'done' });

      // Step 3-5: 후처리
      const result = response.content[0].type === 'tool_use'
        ? response.content[0].input
        : null;

      if (!result) throw new Error('Claude did not return structured result');

      await sendEvent('step', { step: 3, status: 'done' });
      await sendEvent('step', { step: 4, status: 'done' });

      // 이력 저장
      await saveAnalysis(lot, buildingUse, result);

      await sendEvent('step', { step: 5, status: 'done' });
      await sendEvent('result', result);
    } catch (error) {
      await sendEvent('error', { message: error.message });
    } finally {
      await writer.close();
    }
  })();

  return new Response(stream.readable, {
    headers: {
      'Content-Type': 'text/event-stream',
      'Cache-Control': 'no-cache',
      'Connection': 'keep-alive'
    }
  });
}
```

### 4.2 클라이언트 SSE 소비

```typescript
// hooks/useAnalysis.ts
export function useAnalysis() {
  const [step, setStep] = useState(-1);
  const [result, setResult] = useState(null);
  const [error, setError] = useState(null);

  const analyze = async (lot, buildingUse) => {
    setStep(0);
    setResult(null);
    setError(null);

    const response = await fetch('/api/analyze', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ lot, buildingUse })
    });

    const reader = response.body.getReader();
    const decoder = new TextDecoder();
    let buffer = '';

    while (true) {
      const { done, value } = await reader.read();
      if (done) break;

      buffer += decoder.decode(value, { stream: true });
      const lines = buffer.split('\n\n');
      buffer = lines.pop() || '';

      for (const block of lines) {
        const eventMatch = block.match(/event: (\w+)/);
        const dataMatch = block.match(/data: (.+)/);
        if (!eventMatch || !dataMatch) continue;

        const event = eventMatch[1];
        const data = JSON.parse(dataMatch[1]);

        if (event === 'step') setStep(data.step);
        if (event === 'result') setResult(data);
        if (event === 'error') setError(data.message);
      }
    }
  };

  return { step, result, error, analyze };
}
```
